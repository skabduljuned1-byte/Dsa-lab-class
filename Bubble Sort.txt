#include <stdio.h>

void display(int arr[], int size)
{
    for (int k = 0; k < size; k++)
    {
        printf("%d ", arr[k]);
    }
    printf("\n");
}

void bubbleSort(int arr[], int size)
{
    for (int step = 0; step < size - 1; step++)
    {
        int swap = 0;

        for (int i = 0; i < size - step - 1; i++)
        {
            if (arr[i] > arr[i + 1])
            {
                int temp = arr[i];
                arr[i] = arr[i + 1];
                arr[i + 1] = temp;

                swap++;
            }
        }

        printf("Internal comparison %d: ", step + 1);
        display(arr, size);
        printf("\n");

        if (swap == 0)
        {
            break;
        }
    }
}

int main()
{
    int data[] = {6, 4, 7, 8, 1, 9, 5};
    int size = 7;

    bubbleSort(data, size);

    printf("Sorted array: ");
    display(data, size);

    return 0;
}
