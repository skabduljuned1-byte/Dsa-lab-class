#include <stdio.h>

int main() {
    int numbers[5] = {10, 20, 30, 40, 50};

    printf("1st element (index 0): %d\n", numbers[0]);
    printf("Third element (index 2): %d\n", numbers[2]);

    printf("\nAll array elements:\n");

    for (int i = 0; i < 5; i++) {
        printf("Index %d = %d\n", i, numbers[i]);
    }

    return 0;
}
