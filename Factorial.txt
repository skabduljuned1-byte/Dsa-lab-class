#include <stdio.h>

int factorial(int n)
{
    if (n == 0 || n == 1)
    {
        return 1;
    }

    return n * factorial(n - 1);
}

int main()
{
    int a;

    printf("Enter a positive integer: ");
    scanf("%d", &a);

    if (a < 0)
    {
        printf("Factorial is not defined for negative no.\n");
    }
    else
    {
        printf("Factorial of %d is %d\n", a, factorial(a));
    }

    return 0;
}
