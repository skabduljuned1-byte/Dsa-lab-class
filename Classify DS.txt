#include <stdio.h>

int main()
{
    int choice;

    printf("Data structure classification\n");
    printf("1. Array\n");
    printf("2. Linked list\n");
    printf("3. Tree\n");
    printf("4. Graph\n");
    printf("Select a data structure (1-4): ");
    scanf("%d", &choice);

    printf("\nClassification:\n");

    switch (choice)
    {
        case 1:
            printf("Type: Array\n");
            printf("Result = Non primitive, linear and static\n");
            break;

        case 2:
            printf("Type: Linked list\n");
            printf("Result = Non primitive, linear and dynamic\n");
            break;

        case 3:
            printf("Type: Tree\n");
            printf("Result = Non primitive, non linear\n");
            break;

        case 4:
            printf("Type: Graph\n");
            printf("Result = Non primitive, non linear\n");
            break;

        default:
            printf("Invalid choice\n");
    }

    return 0;
}
